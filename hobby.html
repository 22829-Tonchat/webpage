document.addEventListener('DOMContentLoaded', () => { // ฟังก์ชันสร้างหยดน้ำพ่น const createWaterParticle = (e) => { // จำกัดจำนวนตัวเจเนอเรตไม่ให้เครื่องกระตุก const particleCount = 4; for (let i = 0; i < particleCount; i++) { const particle = document.createElement('div'); particle.classList.add('water-particle'); // ตั้งค่าตำแหน่งเริ่มต้นที่หัวลูกศรเมาส์ particle.style.left = `${e.clientX}px`; particle.style.top = `${e.clientY}px`; // สุ่มขนาดหยดน้ำ const size = Math.random() * 8 + 4; // ขนาด 4px - 12px particle.style.width = `${size}px`; particle.style.height = `${size}px`; // สุ่มทิศทางการพ่นกระเด็น (X, Y) และระยะทาง const destinationX = (Math.random() - 0.5) * 80; const destinationY = (Math.random() - 0.5) * 80 - 20; // ให้แรงพ่นเฉียงขึ้นเล็กน้อย particle.style.setProperty('--dx', `${destinationX}px`); particle.style.setProperty('--dy', `${destinationY}px`); // สุ่มระยะเวลาเล่น Animation const duration = Math.random() * 0.4 + 0.4; // 0.4s - 0.8s particle.style.animationDuration = `${duration}s`; document.body.appendChild(particle); // ลบ Element ออกเมื่อทำงานเสร็จเพื่อเคลียร์ Memory setTimeout(() => { particle.remove(); }, duration * 1000); } }; // ดักจับการเคลื่อนไหวของเมาส์ (Throttle เพื่อความลื่นไหล) let isThrottled = false; window.addEventListener('mousemove', (e) => { if (!isThrottled) { createWaterParticle(e); isThrottled = true; setTimeout(() => { isThrottled = false; }, 30); // สร้างทุกๆ 30ms } }); });
